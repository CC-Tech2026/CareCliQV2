# CareCliQ Organization Hierarchy & Data Scoping — Technical Specification

## Executive Summary

This document defines the organizational structure, data hierarchy, role permissions, and data scoping rules for CareCliQ. It ensures:
- ✅ Correct data isolation between organizations
- ✅ Proper role-based access control (RBAC)
- ✅ Secure team/hierarchical data visibility
- ✅ Support for multiple organizational structures
- ✅ Alignment with existing backend architecture

## Database Schema for Hierarchy

### Core Tables Required

```sql
-- Organizations (top level)
CREATE TABLE organizations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name VARCHAR(255) NOT NULL,
  provider_number VARCHAR(50) UNIQUE NOT NULL,
  logo_url VARCHAR(255),
  status VARCHAR(20) NOT NULL DEFAULT 'active', -- active, inactive, archived
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Users (all users, cross-role)
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  organization_id UUID NOT NULL REFERENCES organizations(id),
  email VARCHAR(255) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  first_name VARCHAR(100) NOT NULL,
  last_name VARCHAR(100) NOT NULL,
  role VARCHAR(50) NOT NULL, -- 'support_worker', 'coordinator', 'managing_director', 'admin', 'allied_health'
  
  -- Coordinator field: which coordinator does this worker report to?
  coordinator_id UUID REFERENCES users(id), -- NULL if user is coordinator/MD/admin
  
  -- Status tracking
  status VARCHAR(20) NOT NULL DEFAULT 'active',
  is_active BOOLEAN DEFAULT true,
  last_login TIMESTAMP,
  
  -- Compliance tracking
  compliance_score FLOAT DEFAULT 0,
  
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  -- Constraints
  CONSTRAINT org_isolation CHECK (organization_id IS NOT NULL),
  CONSTRAINT valid_coordinator CHECK (
    (role = 'support_worker' AND coordinator_id IS NOT NULL) OR
    (role IN ('coordinator', 'managing_director', 'admin', 'allied_health') AND coordinator_id IS NULL)
  ),
  CONSTRAINT no_self_coordination CHECK (id != coordinator_id)
);

-- Participants (who receives care)
CREATE TABLE participants (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  organization_id UUID NOT NULL REFERENCES organizations(id),
  first_name VARCHAR(100) NOT NULL,
  last_name VARCHAR(100) NOT NULL,
  ndis_number VARCHAR(50),
  status VARCHAR(20) NOT NULL DEFAULT 'active',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Participant Allocations (which workers serve which participants)
CREATE TABLE participant_allocations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  participant_id UUID NOT NULL REFERENCES participants(id),
  worker_id UUID NOT NULL REFERENCES users(id),
  organization_id UUID NOT NULL REFERENCES organizations(id),
  coordinator_id UUID NOT NULL REFERENCES users(id), -- For quick access to coordinator
  allocation_date DATE NOT NULL,
  status VARCHAR(20) NOT NULL DEFAULT 'active',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  CONSTRAINT valid_worker CHECK (
    -- Verify worker is a support_worker
    EXISTS (SELECT 1 FROM users u WHERE u.id = worker_id AND u.role = 'support_worker')
  ),
  CONSTRAINT valid_coordinator_for_worker CHECK (
    -- Verify coordinator is the worker's coordinator
    EXISTS (
      SELECT 1 FROM users u 
      WHERE u.id = worker_id 
      AND u.coordinator_id = coordinator_id
    )
  )
);

-- Sessions (support delivery evidence)
CREATE TABLE sessions (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  organization_id UUID NOT NULL REFERENCES organizations(id),
  participant_id UUID NOT NULL REFERENCES participants(id),
  worker_id UUID NOT NULL REFERENCES users(id),
  coordinator_id UUID NOT NULL REFERENCES users(id), -- Coordinator for this worker
  session_date DATE NOT NULL,
  start_time TIME NOT NULL,
  end_time TIME NOT NULL,
  duration_minutes INT NOT NULL,
  status VARCHAR(20) NOT NULL DEFAULT 'draft',
  compliance_score FLOAT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  CONSTRAINT org_isolation CHECK (organization_id IS NOT NULL)
);

-- Incidents (safety events)
CREATE TABLE incidents (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  organization_id UUID NOT NULL REFERENCES organizations(id),
  participant_id UUID NOT NULL REFERENCES participants(id),
  worker_id UUID NOT NULL REFERENCES users(id),
  coordinator_id UUID NOT NULL REFERENCES users(id),
  incident_date DATE NOT NULL,
  incident_type VARCHAR(100) NOT NULL,
  status VARCHAR(20) NOT NULL DEFAULT 'open',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Certifications (worker compliance tracking)
CREATE TABLE certifications (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  organization_id UUID NOT NULL REFERENCES organizations(id),
  user_id UUID NOT NULL REFERENCES users(id),
  certification_type VARCHAR(100) NOT NULL, -- 'wwc', 'first_aid', 'rsa', etc.
  issue_date DATE NOT NULL,
  expiry_date DATE NOT NULL,
  status VARCHAR(20) NOT NULL DEFAULT 'valid',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

# Part 2: Role-Based Access Control (RBAC)

## Role Definitions

### 1. Support Worker (frontline staff)
**Responsibilities:**
- Create and document sessions
- Report incidents
- Manage their own credentials/certifications
- View their own compliance score

**Data Access:**
- ✅ Can see: Only their own data
- ✅ Can modify: Only their own sessions (draft status)
- ❌ Cannot see: Other workers' data
- ❌ Cannot see: Team/organization metrics
- ❌ Cannot see: Financial data

**Example Queries:**
```sql
-- Get my sessions
SELECT * FROM sessions 
WHERE worker_id = current_user.id 
AND organization_id = current_user.organization_id

-- Get my participants
SELECT DISTINCT p.* FROM participants p
INNER JOIN participant_allocations pa ON p.id = pa.participant_id
WHERE pa.worker_id = current_user.id
AND p.organization_id = current_user.organization_id

-- Get my certifications
SELECT * FROM certifications
WHERE user_id = current_user.id
AND organization_id = current_user.organization_id
```

---

### 2. Support Coordinator (team lead)
**Responsibilities:**
- Manage team of support workers
- Oversee participant care
- Review and approve sessions
- Track team compliance
- Manage incidents

**Data Access:**
- ✅ Can see: All data for workers assigned to them + participants
- ✅ Can modify: Sessions from their team (approve/flag)
- ✅ Can modify: Incident records for their team
- ❌ Cannot see: Other teams' data (if multi-team organization)
- ❌ Cannot see: Organization-wide metrics (MD's domain)
- ❌ Cannot see: Financial data

**Example Queries:**
```sql
-- Get my team members
SELECT * FROM users
WHERE coordinator_id = current_user.id
AND organization_id = current_user.organization_id
AND role = 'support_worker'

-- Get my team's sessions
SELECT s.* FROM sessions s
WHERE s.coordinator_id = current_user.id
AND s.organization_id = current_user.organization_id
AND s.session_date >= DATE_SUB(NOW(), INTERVAL 30 DAY)

-- Get participants my team serves
SELECT DISTINCT p.* FROM participants p
INNER JOIN participant_allocations pa ON p.id = pa.participant_id
WHERE pa.coordinator_id = current_user.id
AND p.organization_id = current_user.organization_id

-- Get team compliance
SELECT u.id, u.first_name, u.last_name, AVG(s.compliance_score) as avg_compliance
FROM users u
LEFT JOIN sessions s ON u.id = s.worker_id
WHERE u.coordinator_id = current_user.id
AND u.organization_id = current_user.organization_id
GROUP BY u.id

-- Get team certifications expiring
SELECT u.id, u.first_name, c.certification_type, c.expiry_date
FROM users u
INNER JOIN certifications c ON u.id = c.user_id
WHERE u.coordinator_id = current_user.id
AND c.expiry_date BETWEEN NOW() AND DATE_ADD(NOW(), INTERVAL 30 DAY)
AND u.organization_id = current_user.organization_id
```

### 3. Managing Director (organization lead)
**Responsibilities:**
- Oversee entire organization
- Monitor all metrics and trends
- Manage coordinators
- Strategic decision-making
- Financial oversight

**Data Access:**
- ✅ Can see: ALL organization data
- ✅ Can modify: All organization settings
- ✅ Can modify: Staff assignments, role changes
- ✅ Can access: Financial data
- ✅ Can access: Organization-wide metrics

**Example Queries:**
```sql
-- Get organization overview
SELECT 
  (SELECT COUNT(*) FROM participants WHERE organization_id = current_user.organization_id) as active_participants,
  (SELECT COUNT(*) FROM users WHERE organization_id = current_user.organization_id AND is_active = true) as active_staff,
  (SELECT COUNT(*) FROM sessions WHERE organization_id = current_user.organization_id AND session_date >= CURDATE()) as sessions_today,
  (SELECT AVG(compliance_score) FROM users WHERE organization_id = current_user.organization_id) as avg_compliance
  
-- Get all coordinators
SELECT * FROM users
WHERE organization_id = current_user.organization_id
AND role = 'coordinator'

-- Get all incidents
SELECT * FROM incidents
WHERE organization_id = current_user.organization_id
ORDER BY incident_date DESC

-- Get organization compliance trend
SELECT DATE(created_at), AVG(compliance_score)
FROM sessions
WHERE organization_id = current_user.organization_id
GROUP BY DATE(created_at)
ORDER BY DATE(created_at) DESC
```

# Part 3: Data Scoping Implementation

## The Data Scoping Rule

**CRITICAL SECURITY RULE:**
Every single query must filter by BOTH `organization_id` AND role-based visibility.

```
FUNDAMENTAL RULE:
No user can EVER see data outside their organization.
No user can EVER see data they don't have permission to see.
```

## Implementation Pattern

### Pattern 1: Support Worker (Only Own Data)

```typescript
// Support Worker Backend Service
class SupportWorkerService {
  
  // Get my sessions
  async getMyS sessions(userId: string) {
    const user = await this.getUser(userId)
    
    // RULE 1: Only from my organization
    // RULE 2: Only my sessions
    return db.query(`
      SELECT * FROM sessions
      WHERE worker_id = $1
      AND organization_id = $2
    `, [userId, user.organization_id])
  }
  
  // Get my participants
  async getMyParticipants(userId: string) {
    const user = await this.getUser(userId)
    
    return db.query(`
      SELECT DISTINCT p.* FROM participants p
      INNER JOIN participant_allocations pa ON p.id = pa.participant_id
      WHERE pa.worker_id = $1
      AND p.organization_id = $2
    `, [userId, user.organization_id])
  }
  
  // Update session (only draft sessions)
  async updateSession(userId: string, sessionId: string, updates: any) {
    const user = await this.getUser(userId)
    
    // Verify ownership
    const session = await db.query(`
      SELECT * FROM sessions
      WHERE id = $1
      AND worker_id = $2
      AND organization_id = $3
    `, [sessionId, userId, user.organization_id])
    
    if (!session || session.status !== 'draft') {
      throw new UnauthorizedError('Cannot modify this session')
    }
    
    return db.query(`
      UPDATE sessions
      SET ${updateClause}
      WHERE id = $1
      AND worker_id = $2
      AND organization_id = $3
    `, [...updateValues, sessionId, userId, user.organization_id])
  }
}
```

  }

    return db.query(`

      WHERE s.coordinator_id = $1
      AND s.organization_id = $2
      AND s.status IN ('flagged', 'needs_review')
    `, [coordinatorId, user.organization_id])
  }
  
  // Approve/Flag session (only my team's sessions)
  async flagSession(coordinatorId: string, sessionId: string, reason: string) {
    const user = await this.getUser(coordinatorId)
    
    // Verify this session belongs to my team
    const session = await db.query(`
      SELECT * FROM sessions
      WHERE id = $1
      AND coordinator_id = $2
      AND organization_id = $3
    `, [sessionId, coordinatorId, user.organization_id])
    
    if (!session) {
      throw new UnauthorizedError('Cannot access this session')
    }
    
    return db.query(`
      UPDATE sessions
      SET status = 'flagged',
          flagged_reason = $1,
          flagged_at = NOW()
      WHERE id = $2
      AND coordinator_id = $3
      AND organization_id = $4
    `, [reason, sessionId, coordinatorId, user.organization_id])
  }
}

      ORDER BY date DESC

## The Data Scoping Middleware

```typescript
// Critical middleware that enforces data scoping on EVERY request

interface AuthenticatedRequest extends Request {
  user: {
    id: string
    organization_id: string
    role: 'support_worker' | 'coordinator' | 'managing_director' | 'admin'
    coordinator_id?: string // if support_worker
  }
}

// Apply to every API endpoint
async function enforceDat aScoping(req: AuthenticatedRequest, res: Response, next: Function) {
  
  // Get the resource being requested
  const resourceId = req.params.id
  const resourceType = req.baseUrl // /api/v1/sessions, /api/v1/participants, etc.
  
  // Verify organization isolation
  const resource = await getResource(resourceType, resourceId)
  
  if (!resource) {
    return res.status(404).json({ error: 'Not found' })
  }
  
  // CRITICAL CHECK: Organization must match
  if (resource.organization_id !== req.user.organization_id) {
    return res.status(403).json({ error: 'Unauthorized: Access denied' })
  }
  
  // Role-based visibility check
  if (!hasPermission(req.user, resource)) {
    return res.status(403).json({ error: 'Unauthorized: Insufficient permissions' })
  }
  
  next()
}

function hasPermission(user: any, resource: any): boolean {
  
  // Support Worker: Can only access own data
  if (user.role === 'support_worker') {
    if (resource.type === 'session') {
      return resource.worker_id === user.id
    }
    if (resource.type === 'certification') {
      return resource.user_id === user.id
    }
    return false
  }
  
  // Coordinator: Can access team data
  if (user.role === 'coordinator') {
    if (resource.type === 'session') {
      return resource.coordinator_id === user.id
    }
    if (resource.type === 'incident') {
      return resource.coordinator_id === user.id
    }
    if (resource.type === 'participant_allocation') {
      return resource.coordinator_id === user.id
    }
    return false
  }
  
  // MD/Admin: Can access all organization data
  if (user.role === 'managing_director' || user.role === 'admin') {
    return true
  }
  
  return false
}
```
